{% comment %} Compute neighbours over the visible (non-draft) posts only, instead of using Jekyll's built-in page.previous / page.next which walk *all* posts — otherwise a draft sitting between two live posts would leak into their nav. site.posts is newest-first, so the older post is the next index, the newer the previous index. A draft page isn't in the list, so it simply gets no nav. {% endcomment %} {% assign visible_posts = site.posts | where_exp: "p", "p.draft != true" %} {% assign prev_post = nil %} {% assign next_post = nil %} {% for p in visible_posts %} {% if p.url == page.url %} {% unless forloop.last %}{% assign prev_post = visible_posts[forloop.index] %}{% endunless %} {% unless forloop.first %}{% assign newer_idx = forloop.index0 | minus: 1 %}{% assign next_post = visible_posts[newer_idx] %}{% endunless %} {% break %} {% endif %} {% endfor %} {% if prev_post or next_post %}
{% if prev_post %} ← Previous {{ prev_post.title }} {% else %} {% endif %} {% if next_post %} Next → {{ next_post.title }} {% else %} {% endif %}
{% endif %}
